# Phase 10.5 Implementation Plan — Scenario simulator (rent-a-vehicle)

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make the existing 11-agent swarm rent a vehicle end-to-end. A customer types a request at `/chat`, agents check inventory, quote, gate (HITL) the rental, gate the charge, and the operator watches it happen on `/map`, `/billing`, and the bento home.

**Architecture:** Four new domain tables (`vehicles`, `customers`, `trips`, `invoices`) live in `db/schema/domain.ts`. A second in-process MCP server (`auto-ride-domain`) ships eight tools in `lib/mcp/domain.ts`, wired into the orchestrator's `mcpServers` map alongside the existing `memory` server. The eleven agents in `lib/agent/agents/index.ts` are **not** added to — only their `prompt` and `tools` allowlists are extended in-place. Two new gate types (`rental-confirm`, `payment-charge`) reuse phase 07's `approval_gates` row + Inngest `step.waitForEvent` mechanics; only the canUseTool policy in `lib/agent/approval-policy.ts` and the `<ApprovalRequest>` payload renderer learn the two new tool names. Three pages (`/chat`, `/map`, `/billing`) and one bento widget (`trips-on-road`) consume the same data.

**Tech Stack:** TypeScript / Next.js 16 App Router / React 19 / Tailwind 4 / Drizzle ORM 0.45 / Neon Postgres / `maplibre-gl` (mapcn vendored) / `framer-motion` (NEW; vendored deployment-card depends on it) / Vitest 4 / Playwright (existing).

**Spec:** `docs/superpowers/specs/2026-04-28-phase-10.5-scenario-simulator-design.md` — read it first; this plan does not re-explain the design.

**Scope deviation from spec to call out:**
The spec lists 10 domain tools. This plan ships **8** in v0 — `flag_trip_incident` and `audit_approval` are deferred to phase 10.6 because the two seed scenarios (rent-a-vehicle, trip-status) don't exercise them. `incident-responder` and `compliance-auditor` agents stay registered with their phase-08 prompts unchanged. v0 covers the demo path; the 10-tool matrix is the design ceiling.

// db/queries/domain.ts
import { and, eq, gte, sql } from 'drizzle-orm';
import { db } from '@/db/client';
import { customers, invoices, trips, vehicles } from '@/db/schema/domain';

export type QueryVehiclesArgs = {
  status?: 'available' | 'in-use' | 'maintenance';
  segment?: 'sedan' | 'suv' | 'van' | 'truck';
  seats_min?: number;
  range_min_km?: number;
};

export async function queryVehicles(args: QueryVehiclesArgs) {
  const filters = [];
  if (args.status) filters.push(eq(vehicles.status, args.status));
  if (args.segment) filters.push(eq(vehicles.segment, args.segment));
  if (args.seats_min !== undefined) filters.push(gte(vehicles.seats, args.seats_min));
  if (args.range_min_km !== undefined) filters.push(gte(vehicles.rangeKm, args.range_min_km));
  return db
    .select()
    .from(vehicles)
    .where(filters.length ? and(...filters) : undefined);
}

export async function checkVehicleHealth(vehicleId: string) {
  const [row] = await db.select().from(vehicles).where(eq(vehicles.id, vehicleId));
  if (!row) return { ok: false, flags: ['unknown_vehicle'] };
  const flags: string[] = [];
  if (row.status === 'maintenance') flags.push('in_maintenance');
  if (row.lastServicedAt) {
    const days = (Date.now() - row.lastServicedAt.getTime()) / 86_400_000;
    if (days > 90) flags.push('overdue_service');
  }
  return {
    ok: flags.length === 0,
    last_serviced_at: row.lastServicedAt,
    odometer_km: row.odometerKm,
    flags,
  };
}

export async function getTripStatus(args: { trip_id?: string; customer_email?: string }) {
  let trip;
  if (args.trip_id) {
    [trip] = await db.select().from(trips).where(eq(trips.id, args.trip_id));
  } else if (args.customer_email) {
    const [cust] = await db.select().from(customers).where(eq(customers.email, args.customer_email));
    if (!cust) return null;
    [trip] = await db
      .select()
      .from(trips)
      .where(and(eq(trips.customerId, cust.id), sql`${trips.status} IN ('confirmed','in-route')`))
      .limit(1);
  }
  if (!trip) return null;
  const [vehicle] = await db.select().from(vehicles).where(eq(vehicles.id, trip.vehicleId));
  const [customer] = await db.select().from(customers).where(eq(customers.id, trip.customerId));
  return { trip, vehicle, customer };
}

export async function queryCustomerHistory(args: { email?: string; customer_id?: string }) {
  let customer;
  if (args.customer_id) {
    [customer] = await db.select().from(customers).where(eq(customers.id, args.customer_id));
  } else if (args.email) {
    [customer] = await db.select().from(customers).where(eq(customers.email, args.email));
  }
  if (!customer) return null;
  const tripRows = await db.select().from(trips).where(eq(trips.customerId, customer.id));
  const totalSpend = tripRows.reduce((sum, t) => sum + Number(t.totalCostUsd), 0);
  return { customer, trips: tripRows, total_spend_usd: totalSpend };
}

const KM_PER_DEG = 111;
const ROAD_FACTOR = 1.25;

export function computeRoute(args: {
  pickup: { lat: number; lng: number };
  dropoff: { lat: number; lng: number };
}): { distance_km: number; eta_minutes: number; polyline: Array<[number, number]> } {
  const dLat = args.dropoff.lat - args.pickup.lat;
  const dLng = args.dropoff.lng - args.pickup.lng;
  const haversineKm = Math.sqrt(dLat * dLat + dLng * dLng) * KM_PER_DEG;
  const distance_km = Math.round(haversineKm * ROAD_FACTOR * 10) / 10;
  const eta_minutes = Math.round((distance_km / 60) * 60); // assume 60 km/h
  const polyline: Array<[number, number]> = Array.from({ length: 32 }, (_, i) => {
    const t = i / 31;
    return [args.pickup.lat + dLat * t, args.pickup.lng + dLng * t];
  });
  return { distance_km, eta_minutes, polyline };
}

export function quoteRental(args: {
  vehicle: { dailyRateUsd: string; perKmRateUsd: string };
  distance_km: number;
  days: number;
}) {
  const base = Number(args.vehicle.dailyRateUsd) * args.days;
  const perKm = Number(args.vehicle.perKmRateUsd) * args.distance_km;
  const insurance = base * 0.08;
  const subtotal = base + perKm + insurance;
  const service_fee = subtotal * 0.085;
  const total = subtotal + service_fee;
  return {
    base_usd: round2(base),
    per_km_usd: round2(perKm),
    insurance_usd: round2(insurance),
    service_fee_usd: round2(service_fee),
    total_usd: round2(total),
    breakdown: [
      `Base ${args.days}d × $${args.vehicle.dailyRateUsd}/day = $${round2(base)}`,
      `Distance ${args.distance_km} km × $${args.vehicle.perKmRateUsd}/km = $${round2(perKm)}`,
      `Insurance (8%) = $${round2(insurance)}`,
      `Service fee (8.5%) = $${round2(service_fee)}`,
    ],
  };
}

function round2(n: number) {
  return Math.round(n * 100) / 100;
}

        totalCostUsd: args.total_usd.toString(),
        polyline: args.polyline,
        ...(args.task_id ? { taskId: args.task_id } : {}),
      })
      .returning();
    const dueAt = new Date(args.scheduled_start_at);
    dueAt.setDate(dueAt.getDate() - 1);
    const [invoice] = await tx
      .insert(invoices)
      .values({
        tripId: trip.id,
        customerId: args.customer_id,
        amountUsd: (args.total_usd - args.tax_usd).toString(),
        taxUsd: args.tax_usd.toString(),
        totalUsd: args.total_usd.toString(),
        status: 'draft',
        dueAt,
      })
      .returning();
    await tx
      .update(vehicles)
      .set({ status: 'in-use' })
      .where(eq(vehicles.id, args.vehicle_id));
    return { trip, invoice };
  });
}

export async function chargeInvoice(args: { invoice_id: string }) {
  const [invoice] = await db
    .update(invoices)
    .set({ status: 'paid', paidAt: new Date() })
    .where(eq(invoices.id, args.invoice_id))
    .returning();
  if (!invoice) throw new Error(`invoice ${args.invoice_id} not found`);
  return { invoice };
}

// components/vehicles/vehicle-to-deployment.ts
import type { DeploymentData, DeploymentStep } from '@/components/ui/deployment-card';

type Vehicle = {
  id: string; plate: string;
  make: string; model: string; year: number;
  segment: 'sedan'|'suv'|'van'|'truck';
  seats: number; rangeKm: number; odometerKm: number;
  status: 'available'|'in-use'|'maintenance';
  imageSlug: string; depot: string;
  dailyRateUsd: string;
  lastServicedAt: Date | null;
};

type ActiveTrip = {
  id: string;
  distanceKm: string;
  actualStartAt: Date | null;
  scheduledEndAt: Date;
};

export function vehicleToDeploymentData(args: {
  vehicle: Vehicle;
  activeTrip?: ActiveTrip;
  destinationLabel?: string;
  now?: Date;
}): DeploymentData {
  const { vehicle, activeTrip, destinationLabel = 'destination' } = args;
  const now = args.now ?? new Date();
  const status =
    vehicle.status === 'available' ? 'Ready'
    : vehicle.status === 'in-use' ? 'Building'
    : 'Error';

  const lastActive = vehicle.lastServicedAt
    ? `serviced ${Math.max(0, Math.round((now.getTime() - vehicle.lastServicedAt.getTime()) / 86_400_000))}d ago`
    : 'service unknown';

  const idleMessage = `Idle in depot · $${Math.round(Number(vehicle.dailyRateUsd))}/day`;
  const inUseMessage = activeTrip
    ? `Group of ${vehicle.seats} → ${destinationLabel}, ${Math.round(Number(activeTrip.distanceKm))}km`
    : idleMessage;

  const commitMessage = vehicle.status === 'in-use' ? inUseMessage : idleMessage;
  const commitHash = activeTrip ? activeTrip.id.slice(0, 7) : vehicle.id.slice(0, 7);
  const domains = vehicle.status === 'in-use' && activeTrip
    ? [`trip-${activeTrip.id.slice(0, 8)}.in-route`]
    : vehicle.status === 'maintenance'
    ? [`service.bay-1`]
    : [`${vehicle.depot}.idle`];

  const steps = makeSteps({ vehicle, activeTrip, now });

  return {
    id: vehicle.plate,
    environment: vehicle.segment.toUpperCase(),
    status,
    createdTime: `${vehicle.year} · ${vehicle.odometerKm.toLocaleString()} km`,
    createdBy: { name: `${vehicle.make} ${vehicle.model}`, avatar: '' },
    duration: `${vehicle.rangeKm} km · ${vehicle.seats}p`,
    lastActive,
    domains,
    branch: vehicle.depot,
    commitMessage,
    commitHash,
    steps,
    imagePreviewUrl: `/vehicles/${vehicle.imageSlug}.png`,
    imageAlt: `${vehicle.make} ${vehicle.model} ${vehicle.year}`,
  };
}

function makeSteps(args: {
  vehicle: Vehicle;
  activeTrip?: ActiveTrip;
  now: Date;
}): DeploymentStep[] {
  const empty: DeploymentStep = { id: '', label: '', status: 'pending', progress: 0, duration: '—' };
  const labels = ['Pickup', 'In-route', 'Drop-off', 'Return'];
  if (args.vehicle.status === 'maintenance') {
    return [
      { ...empty, id: 'pickup', label: labels[0], status: 'error', duration: 'service' },
      { ...empty, id: 'inroute', label: labels[1] },
      { ...empty, id: 'dropoff', label: labels[2] },
      { ...empty, id: 'return', label: labels[3] },
    ];
  }
  if (args.vehicle.status !== 'in-use' || !args.activeTrip || !args.activeTrip.actualStartAt) {
    return labels.map((label, i) => ({ ...empty, id: `s${i}`, label }));
  }
  const startMs = args.activeTrip.actualStartAt.getTime();
  const endMs = args.activeTrip.scheduledEndAt.getTime();
  const elapsedMs = args.now.getTime() - startMs;
  const totalMs = Math.max(1, endMs - startMs);
  const progress = Math.max(0, Math.min(1, elapsedMs / totalMs));
  return [
    { id: 'pickup', label: 'Pickup', status: 'success', progress: 1, duration: '' },
    { id: 'inroute', label: 'In-route', status: progress >= 1 ? 'success' : 'loading', progress, duration: `${Math.round(progress * 100)}%` },
    { id: 'dropoff', label: 'Drop-off', status: 'pending', progress: 0, duration: '—' },
    { id: 'return', label: 'Return', status: 'pending', progress: 0, duration: '—' },
  ];
}

```tsx
// components/map/vehicle-marker.tsx
'use client';

import { Marker } from '@/components/ui/map';

const COLOR = {
  available: '#22c55e',
  'in-use': '#f59e0b',
  maintenance: '#ef4444',
} as const;

export function VehicleMarker({
  vehicle, activeTrip, onClick,
}: { vehicle: any; activeTrip?: any; onClick: () => void }) {
  const [lat, lng] = activeTrip?.polyline
    ? interpolate(activeTrip.polyline, computeProgress(activeTrip))
    : [vehicle.homeLat, vehicle.homeLng];
  return (
    <Marker latitude={lat} longitude={lng} onClick={onClick}>
      <button
        aria-label={`${vehicle.plate} ${vehicle.status}`}
        className="h-3 w-3 rounded-full border-2 border-white shadow"
        style={{ backgroundColor: COLOR[vehicle.status as keyof typeof COLOR] }}
      />
    </Marker>
  );
}

function computeProgress(trip: { actualStartAt: string | null; scheduledEndAt: string }) {
  if (!trip.actualStartAt) return 0;
  const start = new Date(trip.actualStartAt).getTime();
  const end = new Date(trip.scheduledEndAt).getTime();
  return Math.max(0, Math.min(1, (Date.now() - start) / Math.max(1, end - start)));
}

function interpolate(polyline: Array<[number, number]>, t: number): [number, number] {
  if (polyline.length === 0) return [0, 0];
  const idx = Math.min(polyline.length - 1, Math.floor(t * (polyline.length - 1)));
  return polyline[idx];
}

// app/(dashboard)/chat/actions.ts
'use server';

import { redirect } from 'next/navigation';

export async function createScenarioTask(formData: FormData) {
  const prompt = String(formData.get('prompt') ?? '').trim();
  const customerEmail = String(formData.get('customer_email') ?? '').trim() || undefined;
  if (!prompt) return;
  const res = await fetch(new URL('/api/tasks', process.env.NEXT_PUBLIC_BASE_URL ?? 'http://localhost:3100'), {
    method: 'POST',
    headers: { 'content-type': 'application/json' },
    body: JSON.stringify({ kind: 'scenario', prompt, customer_email: customerEmail }),
  });
  const body = await res.json();
  redirect(`/tasks/${body.taskId}`);
}
```

// app/(dashboard)/billing/page.tsx
import { db } from '@/db/client';
import { invoices, customers, trips, vehicles } from '@/db/schema/domain';
import { desc, eq, sql } from 'drizzle-orm';

export const dynamic = 'force-dynamic';

export default async function BillingPage() {
  const pending = await db.select().from(invoices).where(sql`${invoices.status} IN ('draft','pending')`).orderBy(desc(invoices.dueAt)).limit(20);
  const paid = await db.select().from(invoices).where(eq(invoices.status, 'paid')).orderBy(desc(invoices.paidAt)).limit(20);
  const top = await db.select({
    name: customers.name,
    email: customers.email,
    spend: sql<number>`coalesce(sum(${invoices.totalUsd}::numeric), 0)`,
  }).from(customers).leftJoin(invoices, eq(invoices.customerId, customers.id)).groupBy(customers.id).orderBy(sql`coalesce(sum(${invoices.totalUsd}::numeric), 0) desc`).limit(10);

  return (
    <div className="space-y-12 py-8 px-6">
      <section>
        <h2 className="text-sm font-bold uppercase tracking-wider text-neutral-400">Pending</h2>
        <BillingTable rows={pending} />
      </section>
      <section>
        <h2 className="text-sm font-bold uppercase tracking-wider text-neutral-400">Paid (last 90 days)</h2>
        <BillingTable rows={paid} />
      </section>
      <section>
        <h2 className="text-sm font-bold uppercase tracking-wider text-neutral-400">Top customers</h2>
        <ul>{top.map((c) => <li key={c.email}>{c.name} — ${Number(c.spend).toFixed(2)}</li>)}</ul>
      </section>
    </div>
  );
}

function BillingTable({ rows }: { rows: Array<typeof invoices.$inferSelect> }) {
  return (
    <table className="w-full text-sm">
      <thead><tr><th>Invoice</th><th>Status</th><th>Total</th><th>Due</th></tr></thead>
      <tbody>{rows.map((r) => (
        <tr key={r.id}><td>{r.id.slice(0, 8)}</td><td>{r.status}</td><td>${Number(r.totalUsd).toFixed(2)}</td><td>{r.dueAt.toISOString().slice(0, 10)}</td></tr>
      ))}</tbody>
    </table>
  );
}
